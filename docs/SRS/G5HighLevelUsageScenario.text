\noindent
\textbf{Use Case 001 (\hypertarget{UC-001}{UC-001}): Selecting Properties}
\begin{enumerate}
    \item User launches the RealmForge software.
    \item User selects the "Variations" option.
    \item RealmForge developer helps  the user by explaining the options to the user and collects data on user metrics.
    \item User sets up the variation of the game based on suggestions and criteria from the developer for usability testing.
    \item User finishes setting up by beginning the game.
\end{enumerate}
\textbf{Use Case 002 (\hypertarget{UC-002}{UC-002}): Traversal}
\begin{enumerate}
    \item User enters the game world.
    \item User decides which area of the game world to go to based on the information given for each area.
    \item User selects which area to go to and begins traveling.
    \item User enters a new area and begins an encounter with an enemy (see \hyperlink{UC-003}{UC-003}).
    \item User wins the encounter. 
    \item User begins traveling again (repeats step 2 and below).
\end{enumerate}
\textbf{Use Case 003 (\hypertarget{UC-003}{UC-003}): Combat}
\begin{enumerate}
    \item User enters an encounter with the enemy.
    \item User selects attacks to defeat the enemy.
    \item User repeats step 2 until the enemy is defeated.
    \item User picks and collects rewards
    \item User leaves the encounter.
\end{enumerate}
\textbf{Use Case 004 (\hypertarget{UC-005}{UC-005}): Spending Resources}
\begin{enumerate}
    \item User enters town/city/central hub.
    \item User chooses and selects an available option for purchasing.
    \item User confirms the purchase.
    \item User repeats steps 2 and 3 until the user is satisfied or resources are spent.
    \item User leaves the town/city/central hub.
\end{enumerate}
\textbf{Use Case 005 (\hypertarget{UC-006}{UC-006}): Begin Again}
\begin{enumerate}
    \item User loses the encounter.
    \item User returns to the start screen
    \item User selects the "Variations" option (see UC-001).
    \item User begins the game again.
\end{enumerate}
