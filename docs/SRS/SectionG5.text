\noindent
\textbf{Use Case 001 (\hypertarget{UC-001}{UC-001}): Selecting Properties}
\begin{enumerate}
    \item User launches the RealmForge software.
    \item User selects feature variation configurations.
    \item User decides which features of the game to be changed.
    \item User finishes setting up by beginning the game.
\end{enumerate}
\textbf{Traceability: Refined by \ref{controller-equivalence}, Refined by \ref{controller-marking}, Refined by \ref{controller-default}, Refined by \ref{controller-switching}, Refined by \ref{controller-onlyone},
Refined by \ref{ui-start}, Refined by \ref{ui-end}}
\newline
\noindent
\textbf{Use Case 002 (\hypertarget{UC-002}{UC-002}): Playing a full session}
\begin{enumerate}
    \item User selects to begin a new game session.
    \item User enters the main game world.
    \item User decides which area of the game world to go to explore based on the information given for each area.
    \item User selects which area to go to and begins traveling.
    \item User enters a new area and begins an encounter with an enemie/enemies (see \hyperlink{UC-003}{UC-003}).
    \item User is successful (continue below) or fails in the encounter (see \hyperlink{UC-005}{UC-005}). 
    \item User continues traveling the game world (repeats step 2 through 6).
    \item User completes the win condition for the game session (see \hyperlink{UC-005}{UC-005}).
\end{enumerate}
\textbf{Traceability: Refined by \ref{game-variant-health}, Refined by \ref{game-variant-droprate}, Refined by \ref{gam},}
\newline
\noindent
\textbf{Use Case 003 (\hypertarget{UC-003}{UC-003}): Battling enemies}
\begin{enumerate}
    \item User enters an encounter with the enemie/enemies.
    \item User selects attacks to defeat the enemie/enemies.
    \item User repeats step 2 until the enemie/enemies are defeated.
    \item User collects rewards for successfully completing the encounter. 
    \item User returns to game's world map.
\end{enumerate}
\textbf{Traceability: Refined by \ref{game-variant-health}, Refined by \ref{game-variant-droprate}, Refined by \ref{gam},}
\newline
\noindent
\textbf{Use Case 004 (\hypertarget{UC-004}{UC-004}): Upgrading and purchasing items}
\begin{enumerate}
    \item User enters town/city/central hub.
    \item User chooses and selects an available option(s) for purchasing.
    \item User confirms the purchase.
    \item User repeats steps 2 and 3 until the user is satisfied or resources are spent.
    \item User leaves the town/city/central hub to continue the game session.
\end{enumerate}
\textbf{Traceability: Refined by \ref{game-variant-health}, Refined by \ref{game-variant-droprate}, Refined by \ref{gam},}
\newline
\noindent
\textbf{Use Case 005 (\hypertarget{UC-005}{UC-005}): Continuing or start a new game}
\begin{enumerate}
    \item User completes a game session or fails in between (see \hyperlink{UC-002}{UC-002}).
    \item User is given an option to return to the central hub (see \hyperlink{UC-004}{UC-004}) or quit the current game session (see below). 
    \item User selects to change features (see \hyperlink{UC-001}{UC-001}) or proceeds with the same configurations.
    \item User begins a new game.
\end{enumerate}
\textbf{Traceability: Refined by \ref{game-variant-health}, Refined by \ref{game-variant-droprate}, Refined by \ref{gam},}
